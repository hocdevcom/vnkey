find_package(Fcitx5Core REQUIRED)
find_package(Fcitx5Utils REQUIRED)
set(VIET_FCITX_DEFAULT_DIR "${CMAKE_INSTALL_LIBDIR}/fcitx5")
# The host framework may use lib64 or a Debian multiarch directory. Respect it
# for system installs; custom prefixes (including Nix) keep their own tree.
if(CMAKE_INSTALL_PREFIX STREQUAL "/usr")
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_get_variable(VIET_FCITX_NATIVE_LIBDIR Fcitx5Core libdir)
        if(VIET_FCITX_NATIVE_LIBDIR)
            set(VIET_FCITX_DEFAULT_DIR "${VIET_FCITX_NATIVE_LIBDIR}/fcitx5")
        endif()
    endif()
endif()
set(VIET_FCITX_ADDON_DIR "${VIET_FCITX_DEFAULT_DIR}" CACHE PATH "Fcitx5 native addon installation directory")
add_library(viet MODULE engine.cpp)
target_compile_features(viet PRIVATE cxx_std_17)
target_compile_options(viet PRIVATE -Wall -Wextra)
if(VIET_WARNINGS_AS_ERRORS)
    target_compile_options(viet PRIVATE -Werror)
endif()
target_link_libraries(viet PRIVATE VietCore Fcitx5::Core Fcitx5::Utils)
get_filename_component(VIET_CORE_ABSOLUTE_DIR "${VIET_CORE_INSTALL_DIR}" ABSOLUTE BASE_DIR "${CMAKE_INSTALL_PREFIX}")
get_filename_component(VIET_FCITX_ABSOLUTE_DIR "${VIET_FCITX_ADDON_DIR}" ABSOLUTE BASE_DIR "${CMAKE_INSTALL_PREFIX}")
file(RELATIVE_PATH VIET_FCITX_CORE_RELATIVE "${VIET_FCITX_ABSOLUTE_DIR}" "${VIET_CORE_ABSOLUTE_DIR}")
set_target_properties(viet PROPERTIES INSTALL_RPATH "$ORIGIN/${VIET_FCITX_CORE_RELATIVE}")
install(TARGETS viet LIBRARY DESTINATION "${VIET_FCITX_ADDON_DIR}")
install(FILES viet-addon.conf DESTINATION "${CMAKE_INSTALL_DATADIR}/fcitx5/addon" RENAME viet.conf)
install(FILES viet.conf DESTINATION "${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod")
