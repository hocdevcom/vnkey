find_package(PkgConfig REQUIRED)
pkg_check_modules(IBUS REQUIRED IMPORTED_TARGET ibus-1.0>=1.5.27)
add_executable(ibus-viet-plugin engine.c)
target_compile_features(ibus-viet-plugin PRIVATE c_std_11)
target_compile_options(ibus-viet-plugin PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ibus-viet-plugin PRIVATE VietCore PkgConfig::IBUS)
set_target_properties(ibus-viet-plugin PROPERTIES INSTALL_RPATH "${CMAKE_INSTALL_FULL_LIBDIR}")
configure_file(viet.xml.in viet.xml @ONLY)
install(TARGETS ibus-viet-plugin RUNTIME DESTINATION "${CMAKE_INSTALL_LIBEXECDIR}/viet-ime")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/viet.xml" DESTINATION "${CMAKE_INSTALL_DATADIR}/ibus/component")
