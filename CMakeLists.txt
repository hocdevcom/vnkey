cmake_minimum_required(VERSION 3.16)
project(viet-ime VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build the framework wrappers on Linux; core-rust also builds standalone on Windows")
endif()
option(VIET_BUILD_IBUS "Build IBus transport adapter" ON)
option(VIET_BUILD_FCITX5 "Build Fcitx5 transport adapter" ON)
option(VIET_WARNINGS_AS_ERRORS "Treat adapter compiler warnings as errors" OFF)
set(VIET_CORE_INSTALL_DIR "${CMAKE_INSTALL_LIBDIR}/viet-ime" CACHE STRING "Private Rust library directory")
find_program(CARGO_EXECUTABLE cargo)
if(NOT CARGO_EXECUTABLE)
    message(FATAL_ERROR "cargo is required to build viet-core")
endif()
set(VIET_CORE_LIBRARY "${PROJECT_SOURCE_DIR}/target/release/libviet_core.so")
add_custom_target(viet_core_build ALL
    COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${PROJECT_SOURCE_DIR}/target"
            ${CARGO_EXECUTABLE} build --release --locked --manifest-path "${PROJECT_SOURCE_DIR}/core-rust/Cargo.toml"
    BYPRODUCTS "${VIET_CORE_LIBRARY}"
               "${PROJECT_SOURCE_DIR}/target/release/viet-config"
    WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
    VERBATIM)
add_library(VietCore SHARED IMPORTED GLOBAL)
set_target_properties(VietCore PROPERTIES IMPORTED_LOCATION "${VIET_CORE_LIBRARY}"
    IMPORTED_NO_SONAME TRUE
    INTERFACE_INCLUDE_DIRECTORIES "${PROJECT_SOURCE_DIR}/core-rust/include")
add_dependencies(VietCore viet_core_build)
install(FILES "${VIET_CORE_LIBRARY}" DESTINATION "${VIET_CORE_INSTALL_DIR}")
install(PROGRAMS "${PROJECT_SOURCE_DIR}/target/release/viet-config" DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES packaging/viet-config.desktop DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
install(FILES core-rust/include/viet_core.h DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(FILES LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/viet-ime")
install(FILES core-rust/config.example.toml README.md DESTINATION "${CMAKE_INSTALL_DATADIR}/viet-ime")
if(VIET_BUILD_IBUS)
    add_subdirectory(wrappers/ibus-c)
endif()
if(VIET_BUILD_FCITX5)
    add_subdirectory(wrappers/fcitx5-cpp)
endif()
set(CPACK_PACKAGE_NAME "viet-ime")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Vietnamese input with a shared Rust core")
set(CPACK_PACKAGE_CONTACT "viet-ime contributors")
set(CPACK_RESOURCE_FILE_LICENSE "${PROJECT_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGING_INSTALL_PREFIX "${CMAKE_INSTALL_PREFIX}")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(VIET_RUNTIME_DEPENDS zenity)
if(VIET_BUILD_IBUS)
    list(APPEND VIET_RUNTIME_DEPENDS ibus)
endif()
if(VIET_BUILD_FCITX5)
    list(APPEND VIET_RUNTIME_DEPENDS fcitx5)
    set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "fcitx5-frontend-all, fcitx5-config-qt")
endif()
string(REPLACE ";" ", " VIET_RUNTIME_DEPENDS "${VIET_RUNTIME_DEPENDS}")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "${VIET_RUNTIME_DEPENDS}")
set(CPACK_RPM_PACKAGE_AUTOREQ ON)
set(CPACK_RPM_PACKAGE_REQUIRES "${VIET_RUNTIME_DEPENDS}")
set(CPACK_RPM_PACKAGE_LICENSE "MIT")
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION
    /usr/lib /usr/lib64 /usr/libexec /usr/share/ibus /usr/share/ibus/component
    /usr/share/fcitx5 /usr/share/fcitx5/addon /usr/share/fcitx5/inputmethod)
include(CPack)
